seurat_objects <- lapply(names(folders), function(name) {
  data <- Read10X_h5(folders[[name]]) 
  seurat_obj <- CreateSeuratObject(counts = data,min.cells = 3,min.features = 200)  
  group <- ifelse(grepl("Ctrl", name), "Control", "P20")
  seurat_obj$group <- group
  seurat_obj$sample <- name
  return(seurat_obj)
})
