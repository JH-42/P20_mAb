Tcs<-readRDS("./Tcs_cytotrace2.RDS")


#############################################################Tcs



lym<-Tcs
seurat_data <- as.data.frame(lym@assays$RNA@data)

phate_data_input <- t(seurat_data)

phate_output <- phate(phate_data_input,gamma=-1,t=120) 
phate_output$embedding<-phate_output$embedding*100


lym[["PHATE"]] <- CreateDimReducObject(embeddings = phate_output$embedding, key = "PHATE_", assay = DefaultAssay(lym))
DimPlot(lym,reduction = "PHATE",group.by = "group")+scale_color_manual(values = c("#5F559B99","#BB002199"))
ggsave(filename = "Tcs_group_PHATE_P20.pdf",width=7,height=5.5)


DimPlot(lym,reduction = "PHATE",label = F,group.by = "new.cell",split.by = "group")+scale_color_manual(values = c("#a4a839","#3e3d1a","#e94f9e","#705da9","#231d65","#d6a0c9",
                                                                                                                  "#c8ba8b"))
ggsave(filename = "Tcs_cells_PHATE_P20vsCtrl.pdf",width=9.5,height=4.8)



FeaturePlot(lym,features = "CytoTRACE2_Score",split.by = "group",reduction = "PHATE",raster=NULL,cols = c("navy","green3","yellow"))
ggsave(filename = "Tcs_cytotrace_P20vsCtrl.pdf",width=9.5,height=4.8)

DimPlot(lym,group.by = "CytoTRACE2_Potency",split.by = "group",reduction = "PHATE")+scale_color_d3()
ggsave(filename = "Tcs_cytotrace_potency_P20vsCtrl.pdf",width=9.5,height=4.8)

cell.prop<- as.data.frame(prop.table(table(lym@meta.data$CytoTRACE2_Potency,lym@meta.data$group)))
# cell.prop$Freq<-cell.prop$Freq*100 
colnames(cell.prop)<-c("clusters","type","Percentages")

cell.prop$type <- factor(cell.prop$type, levels = c( "Control","P20"))


ct1<-ggplot(cell.prop,aes(type,Percentages,fill=clusters))+
  geom_bar(stat = "identity",position="fill")+
  ggtitle("")+
  theme_bw()+
  theme(axis.ticks.length=unit(0.08,"cm"))+
  guides(fill=guide_legend(title=NULL))+scale_color_manual(values = c("#5F559B99","#BB002199"))
ggsave(filename="Fig.2E.pdf",ct1,width=3.5,height=5)
export::graph2ppt(file="Tcs_phate_proporation.pptx",ct1,width=3.4,height=6)
