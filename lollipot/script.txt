library(maftools)

laml_mc3 = tcgaLoad(study = c("LUAD","BRCA","COAD","PAAD","PRAD"))

tp53_mafs <- lapply(laml_mc3, function(maf) {
  maf_data <- maf@data
  
  maf_data <- as.data.frame(lapply(maf_data, function(col) {
    if (is.character(col)) {
      Encoding(col) <- "UTF-8"
      col <- iconv(col, from = "UTF-8", to = "UTF-8", sub = "")
    }
    return(col)
  }))
  
  new_maf <- read.maf(maf_data)
  
  subsetMaf(maf = new_maf, genes = "TP53")
})

merged_maf <- merge_mafs(tp53_mafs)



tp53_mutations <- subsetMaf(maf = merged_maf, genes = "TP53")


mutation_frequency <- as.data.frame(table(tp53_mutations@data$HGVSp_Short))
colnames(mutation_frequency) <- c("Mutation", "Frequency")


high_frequency_mutations <- mutation_frequency[mutation_frequency$Frequency > 10, ]
annotated_mutations <- high_frequency_mutations
annotated_mutations$Label <- paste0(annotated_mutations$Mutation, " (", annotated_mutations$Frequency, ")")

top10_mutations <- mutation_frequency[order(-mutation_frequency$Frequency), ][1:10, ]
print(top10_mutations)
